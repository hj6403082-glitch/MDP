# --- Core distributed engine (Phase 1-3) ---
pyspark==3.5.3
# Python 3.12 removed the stdlib `distutils`; pyspark.ml still imports it, so
# setuptools (which provides the distutils shim) must be present. A bare venv on
# 3.12 does not include it by default.
setuptools<81

# --- Numerics / IO shared across phases ---
# numpy relaxed so pip resolves a wheel for the host Python (1.26.4 had no
# build for 3.13/3.14). 2.3+ is compatible with pyspark 3.5, pandas 2.2+,
# xgboost 2.1 and shap 0.46.
numpy>=2.3.0,<3
pandas==2.2.2
pyarrow==16.1.0          # required for Spark<->pandas Arrow conversions + Parquet

# NOTE: this file targets the Dataproc runtime (Python 3.12). To run
# data_collection.py locally on a newer Python (3.13/3.14) where the compiled
# pins below (pandas/pyarrow/scikit-learn/shap/matplotlib) have no wheels yet,
# install requirements-datacollection.txt instead - it has only what that
# script needs, with relaxed versions that resolve on new interpreters.

# --- Raw data collection (data_collection.py; local ETL, not needed on the
#     Dataproc runtime image) ---
requests==2.32.3         # OpenAQ / Open-Meteo REST calls
google-cloud-storage==2.18.2  # upload staged Parquet to the Dataproc bucket

# --- Phase 2a: distributed predictive baseline ---
xgboost==2.1.1           # provides xgboost.spark.SparkXGBRegressor

# --- Phase 2b: causal discovery ---
lingam==1.9.0            # DirectLiNGAM
scikit-learn==1.5.1      # required by lingam (and shap)

# --- Phase 3a: multi-objective optimization ---
optuna==4.0.0            # NSGAIISampler + relational-DB study backend

# --- Phase 3b: explainability (XAI) ---
shap==0.46.0
matplotlib==3.9.2        # headless (Agg) rendering of the SHAP summary plot
